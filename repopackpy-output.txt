"""Reuse validated 50k experiment chunks in separate 100k protocol directories.

Run from the project root AFTER defining *_100k protocols and sweeps in
config/experiment.yml:
    python migrate_50k_to_100k.py
"""

from pathlib import Path
from shutil import copy2

import pandas as pd
import yaml

from config import PROJECT_ROOT, sweep_experiments
from script.experiment.design_matrix import ensure_design_matrix


SWEEPS = [
    "FullGrey224_100k_sweep",
    "FullWhite224_100k_sweep",
    "FullBlack224_100k_sweep",
    "FullAdjacent_100k_sweep",
]


def main():
    checked_designs = set()
    copied_total = 0

    for sweep_id in SWEEPS:
        for experiment in sweep_experiments(sweep_id):
            imageset = experiment["imageset"]
            subset = experiment["subset"]
            design = experiment["trial_design"]
            design_key = (imageset, subset)

            if design_key not in checked_designs:
                matrix_name = (
                    f"{imageset}__{subset}__{design['n_items_per_trial']}items__"
                    f"50000trials__seed{design['sampling_seed']}.csv"
                )
                old_design_path = PROJECT_ROOT / "data" / "design_matrix" / matrix_name
                old_design = pd.read_csv(old_design_path, dtype=str)
                new_design = ensure_design_matrix(experiment).astype(str)
                if not old_design.equals(new_design.iloc[:len(old_design)].reset_index(drop=True)):
                    raise ValueError(f"Design matrix prefix mismatch: {imageset} / {subset}")
                checked_designs.add(design_key)
                print(f"Verified first {len(old_design):,} trials: {imageset} / {subset}")

            old_protocol = experiment["protocol_id"].removesuffix("_100k")
            experiment_id = experiment["experiment_id"]
            old_dir = PROJECT_ROOT / "results" / old_protocol / experiment_id
            new_dir = PROJECT_ROOT / "results" / experiment["protocol_id"] / experiment_id
            old_chunks = old_dir / "chunks"

            if not old_chunks.exists():
                continue

            snapshot = yaml.safe_load((old_dir / "config_snapshot.yml").read_text(encoding="utf-8"))
            expected_50k = {**design, "n_trials": 50_000}
            if (
                snapshot["trial_design"] != expected_50k
                or snapshot["layout"] != experiment["layout"]
                or snapshot["metrics"] != experiment["metrics"]
            ):
                raise ValueError(f"Existing results are incompatible: {old_dir}")

            destination = new_dir / "chunks"
            destination.mkdir(parents=True, exist_ok=True)
            n_copied = 0
            for source in old_chunks.glob("chunk_*_*.csv"):
                target = destination / source.name
                if not target.exists():
                    copy2(source, target)
                    n_copied += 1
            copied_total += n_copied
            print(f"{experiment['protocol_id']} / {experiment_id}: copied {n_copied} chunks")

    print(f"Done: copied {copied_total} chunks. Do not copy old results.csv or config_snapshot.yml.")


if __name__ == "__main__":
    main()

def resolve_protocol(protocol_id: str) -> dict:
    """Resolve a protocol preset into its component configurations."""
    protocol = PROTOCOL_PRESETS[protocol_id]

    layout_id = protocol["layout"]
    trial_design_id = protocol["trial_design"]
    metrics_id = protocol["metrics"]

    return {
        "protocol_id": protocol_id,

        "layout_id": layout_id,
        "layout": deepcopy(LAYOUTS[layout_id]),

        "trial_design_id": trial_design_id,
        "trial_design": deepcopy(TRIAL_DESIGNS[trial_design_id]),

        "metrics_id": metrics_id,
        "metrics": deepcopy(METRICS[metrics_id]),
    }

def sweep_experiments(sweep_id: str):
    """Yield all experiments defined by a sweep."""
    sweep = EXPERIMENT_SWEEPS[sweep_id]
    protocol = resolve_protocol(sweep["protocol"])

    for model_id in sweep["models"]:
        for imageset_id, subsets in sweep["imagesets"].items():
            for subset in subsets:

                experiment_id = (
                    f"{model_id}__{imageset_id}__{subset}"
                )

                yield {
                    "experiment_id": experiment_id,
                    "sweep_id": sweep_id,

                    "model": model_id,
                    "imageset": imageset_id,
                    "subset": subset,

                    **deepcopy(protocol),
                }

# Aggregate trial priority by item and attach ratings and visual features.

read_item_covariates <- function(imageset, root) {
  stimulus_dir <- file.path(root, "stimuli", imageset)
  ratings <- data.table::fread(
    file.path(stimulus_dir, "ratings.csv"),
    select = c("Image_id", "Arousal", "Valence"),
    colClasses = list(character = "Image_id")
  )
  visual <- data.table::fread(
    file.path(stimulus_dir, "visual_features.csv"),
    colClasses = list(character = "Image_id")
  )
  list(ratings = ratings, visual = visual)
}

make_item_priority <- function(trial, covariates) {
  item <- trial[, .(
    Priority = mean(Priority, na.rm = TRUE),
    N_presentations = .N,
    N_valid = sum(!is.na(Priority))
  ), by = .(Image_id, Metric, Variant_id)]

  item <- merge(item, covariates$ratings, by = "Image_id", all.x = TRUE, sort = FALSE)
  merge(item, covariates$visual, by = "Image_id", all.x = TRUE, sort = FALSE)
}

# Fit item-level regressions and correct affective predictor p-values.

fit_priority_regression <- function(item, predictors) {
  needed <- c("Priority", predictors)
  data <- item[stats::complete.cases(item[, ..needed]), ..needed]
  data[, (needed) := lapply(.SD, function(x) as.numeric(scale(x))), .SDcols = needed]

  fit <- stats::lm(
    stats::reformulate(predictors, response = "Priority"),
    data = data
  )
  s <- summary(fit)
  terms <- predictors
  estimates <- s$coefficients[terms, , drop = FALSE]
  intervals <- stats::confint(fit, parm = terms)

  coefficients <- data.table::data.table(
    Predictor = terms,
    Beta = estimates[, "Estimate"],
    SE = estimates[, "Std. Error"],
    CI_low = intervals[, 1],
    CI_high = intervals[, 2],
    P_raw = estimates[, "Pr(>|t|)"]
  )
  statistics <- data.table::data.table(
    N = stats::nobs(fit),
    R2 = s$r.squared,
    Adjusted_R2 = s$adj.r.squared
  )
  list(coefficients = coefficients, statistics = statistics)
}

correct_pvalues <- function(coefficients, config) {
  coefficients[, `:=`(
    P_adjusted = NA_real_,
    Significant = NA,
    Correction_family = NA_character_,
    N_tests_in_family = NA_integer_
  )]

  scope <- unlist(config$scope, use.names = FALSE)
  targets <- unlist(config$predictors, use.names = FALSE)
  indices <- which(coefficients$Predictor %in% targets & !is.na(coefficients$P_raw))

  coefficients[indices, `:=`(
    P_adjusted = stats::p.adjust(P_raw, method = config$method),
    N_tests_in_family = .N,
    Correction_family = paste(unlist(.BY, use.names = FALSE), collapse = "__")
  ), by = scope]

  coefficients[, Significant := !is.na(P_adjusted) & P_adjusted < config$alpha]
  coefficients[is.na(P_adjusted), Significant := NA]
  coefficients[]
}

  }

  list(training = parts[1], size = parts[2], seed = NA_integer_)
}

# Join trial results to their image IDs, then compute priority.

read_trial_results <- function(experiment, root) {
  design <- experiment$trial_design
  matrix_id <- sprintf(
    "%s__%s__%sitems__%strials__seed%s",
    experiment$imageset, experiment$subset,
    design$n_items_per_trial, design$n_trials, design$sampling_seed
  )

  matrix_path <- file.path(root, "data", "design_matrix", paste0(matrix_id, ".csv"))
  result_path <- file.path(root, "results", experiment$protocol, experiment$id, "results.csv")

  matrix <- data.table::fread(matrix_path, colClasses = list(character = c("Constituent_0", "Constituent_1")))
  results <- data.table::fread(result_path)

  positions <- data.table::melt(
    matrix,
    id.vars = "Trial_id",
    measure.vars = grep("^Constituent_", names(matrix), value = TRUE),
    variable.name = "Constituent",
    value.name = "Image_id"
  )
  positions[, Position_index := as.integer(sub("Constituent_", "", Constituent))]

  results <- merge(
    results,
    positions[, .(Trial_id, Position_index, Image_id)],
    by = c("Trial_id", "Position_index"),
    sort = FALSE
  )
  results[, Image_id := as.character(Image_id)]
  results
}

make_trial_priority <- function(results, mode) {
  trial <- data.table::copy(results)

  if (mode == "absolute") {
    trial[, Priority := Value]
  } else if (mode == "relative") {
    trial[, Total := sum(Value), by = .(Trial_id, Metric, Variant_id)]
    trial[, Priority := data.table::fifelse(Total == 0, NA_real_, Value / Total)]
    trial[, Total := NULL]
  } else {
    stop("Unknown priority mode: ", mode)
  }

  data.table::setcolorder(trial, c("Trial_id", "Position_index", "Image_id", "Metric", "Variant_id", "Value", "Priority"))
  trial
}

# %% Imports

# %% Generation
def generate_design_matrix(
    image_ids: list[str],
    n_trials: int,
    n_items_per_trial: int,
    sampling_seed: int,
) -> pd.DataFrame:
    """Generate unique ordered samples in random order."""
    max_trials = perm(len(image_ids), n_items_per_trial)

    if n_trials > max_trials:
        raise ValueError(
            f"Requested {n_trials} trials, but only {max_trials} unique trials possible."
        )
        
    rng = random.Random(sampling_seed)
    trials = []
    seen = set()

    while len(trials) < n_trials:
        trial = tuple(rng.sample(image_ids, n_items_per_trial))
        if trial not in seen:
            seen.add(trial)
            trials.append(trial)

    columns = [f"Constituent_{i}" for i in range(n_items_per_trial)]
    df = pd.DataFrame(trials, columns=columns)
    df.insert(0, "Trial_id", range(len(df)))
    return df

def ensure_design_matrix(
    experiment: dict,
    overwrite: bool = False,
) -> pd.DataFrame:
    """Load or generate the design matrix for an experiment."""
    imageset = experiment["imageset"]
    subset = experiment["subset"]
    design = experiment["trial_design"]

    matrix_id = (
        f"{imageset}__{subset}__"
        f"{design['n_items_per_trial']}items__"
        f"{design['n_trials']}trials__seed{design['sampling_seed']}"
    )
    path = PROJECT_ROOT / "data" / "design_matrix" / f"{matrix_id}.csv"

    if path.exists() and not overwrite:
        message(f"Loading design matrix: {path}")
        return pd.read_csv(path)

    df = generate_design_matrix(get_image_ids(imageset, subset), **design)
    path.parent.mkdir(parents=True, exist_ok=True)
    df.to_csv(path, index=False)
    message(f"Generated design matrix: {path}")

def _format_chunk_path(chunk_dir: Path, start: int, end: int) -> Path:
    """Build a chunk path for the trial interval."""
    fmt = lambda i: f"{i//1000}k" if i >= 1000 and i % 1000 == 0 else str(i)
    return chunk_dir / f"chunk_{fmt(start)}_{fmt(end)}.csv"

def _save_chunk(chunk_results: list[pd.DataFrame], path: Path):
    """Save a chunk."""
    chunk_df = pd.concat(chunk_results, ignore_index=True)
    temp_path = path.with_suffix(".tmp")
    chunk_df.to_csv(temp_path, index=False)
    temp_path.replace(path)

def _parse_chunk_start(path: Path) -> int:
    """Parse the starting trial index from a chunk filename."""
    start = path.stem.split("_")[1]
    return int(start[:-1]) * 1000 if start.endswith("k") else int(start)

def _merge_chunks(chunk_dir: Path, result_path: Path):
    """Merge chunks to get results.csv."""
    temp_path = result_path.with_suffix(".tmp")
    with open(temp_path, "w", encoding="utf-8") as f:
        for i, path in enumerate(sorted(chunk_dir.glob("chunk_*_*.csv"), key=_parse_chunk_start)):
            pd.read_csv(path).to_csv(f, index=False, header=(i == 0))
    temp_path.replace(result_path)

# Process trials in batches to limit memory usage.
def _prepare_batch_inputs(
    batch_design_matrix: pd.DataFrame,
    image_cache: dict,
    constituent_embedding_cache: dict,
    layout: dict,
    device: torch.device,
) -> tuple[torch.Tensor, dict[str, list[torch.Tensor]]]:
    """Prepare composite images and constituent embeddings for one batch."""
    constituent_columns = [
        col for col in batch_design_matrix
        if col.startswith("Constituent_")
    ]

    constituent_images = []
    embeddings_by_variant = {
        config_key: []
        for config_key in constituent_embedding_cache
    }

    for position_index, column in enumerate(constituent_columns):
        image_ids = batch_design_matrix[column]

        images = torch.stack([
            image_cache[image_id]
            for image_id in image_ids
        ]).to(device=device, dtype=torch.float32) / 255

        constituent_images.append(images)

        for config_key, cached_embeddings in constituent_embedding_cache.items():
            embeddings = torch.stack([
                cached_embeddings[image_id][position_index]
                for image_id in image_ids
            ])
            embeddings_by_variant[config_key].append(embeddings)

    composites = make_composite(constituent_images, layout)
    return composites, embeddings_by_variant


def _process_batch(
    batch_design_matrix: pd.DataFrame,
    experiment: dict,
    model,
    image_cache: dict,
    constituent_embedding_cache: dict,
    attention_cache: list,
    constituent_patch_indices: list[list[int]],
) -> pd.DataFrame:
    """Process one batch and return long-form metric results."""
    device = next(model.parameters()).device

    composites, constituent_embeddings_by_variant = _prepare_batch_inputs(
        batch_design_matrix,
        image_cache,
        constituent_embedding_cache,
        experiment["layout"],
        device,
    )

    composite_outputs = forward_composites(
        model, composites, experiment["metrics"], attention_cache
    )

    return compute_metrics(
        batch_design_matrix["Trial_id"],
        composite_outputs,
        experiment["metrics"],
        constituent_embeddings_by_variant,
        constituent_patch_indices,
    )


# %% Chunk execution
# Group batches into chunks for periodic saving and resumption.
def _run_by_chunks(
    experiment: dict,
    design_matrix: pd.DataFrame,
    model,
    image_cache: dict,
    constituent_embeddings_by_variant: dict,
    attention_cache: list,
    constituent_patch_indices: list[list[int]],
    batch_size: int,
    chunk_dir: Path,
):
    """Process and save experiment results in resumable chunks."""
    with tqdm(
        total=len(design_matrix),
        desc=experiment["experiment_id"],
        unit="trial",
        position=1,
        leave=False,
    ) as progress:
        for chunk_start in range(0, len(design_matrix), CHUNK_SIZE):
            chunk_end = min(chunk_start + CHUNK_SIZE, len(design_matrix))
            path = _format_chunk_path(chunk_dir, chunk_start, chunk_end)

            if path.exists():
                progress.update(chunk_end - chunk_start)
                continue

            chunk_results = []
            for start in range(chunk_start, chunk_end, batch_size):
                batch_design_matrix = design_matrix.iloc[
                    start:min(start + batch_size, chunk_end)
                ]
                batch_results = _process_batch(
                    batch_design_matrix,
                    experiment,
                    model,
                    image_cache,
                    constituent_embeddings_by_variant,
                    attention_cache,
                    constituent_patch_indices,
                )
                chunk_results.append(batch_results)
                progress.update(len(batch_design_matrix))

            _save_chunk(chunk_results, path)

    imageset = experiment["imageset"]
    subset = experiment["subset"]

    result_dir, chunk_dir, result_path = _prepare_results(experiment)
    if result_path.exists():
        message(f"Already completed: {experiment['experiment_id']}")
        return result_dir

    design_matrix = ensure_design_matrix(experiment)
    device = get_device()
    model, batch_size, constituent_embeddings_by_variant = (
        _prepare_model_and_embeddings(experiment, device)
    )

    message(f"Experiment: {experiment['experiment_id']}")
    message(f"Device: {device} | Batch size: {batch_size}")

    image_cache = load_image_cache(imageset, subset)
    attention_cache, handles, constituent_patch_indices = setup_attention(
        model, experiment["metrics"], experiment["layout"]
    )

    try:
        _run_by_chunks(

            model,
            image_cache,
            constituent_embeddings_by_variant,
            attention_cache,
            constituent_patch_indices,
            batch_size,
            chunk_dir,
        )
    finally:
        remove_hooks(handles)

    _merge_chunks(chunk_dir, result_path)
    message(f"Results saved: {result_path}")
    return result_dir

"""
Compute metrics from composite embeddings and attention matrices.
"""

# %% Imports
import pandas as pd
import torch
import torch.nn.functional as F


# %% Attention computation
# Raw attention dimensions: [Batch, Layer, Head, Token, Token]
def _aggregate_heads(attention: torch.Tensor, method: str):
    """Aggregate attention across heads."""
    if method == "mean":
        return attention.mean(dim=-3)
    if method == "max":
        return attention.amax(dim=-3)
    if method == "min":
        return attention.amin(dim=-3)


def _discard_attention(attention: torch.Tensor, ratio: float):
    """Discard the lowest attention values."""
    flat = attention.flatten(start_dim=2)
    n_discard = int(flat.shape[-1] * ratio)

    if n_discard:
        indices = flat.argsort(dim=-1)[..., :n_discard]
        flat.scatter_(-1, indices, 0)

    return flat.reshape_as(attention)


def _add_residual_and_normalize(attention: torch.Tensor):
    """Add residual connections and normalize attention rows."""
    n_tokens = attention.shape[-1]
    identity = torch.eye(
        n_tokens, device=attention.device, dtype=attention.dtype
    ).view(1, 1, n_tokens, n_tokens)

    attention = attention + identity
    return attention / attention.sum(dim=-1, keepdim=True)


def attention_rollout(attentions: torch.Tensor, config: dict):
    """Compute attention rollout across transformer layers."""
    attention = _aggregate_heads(attentions, config["head_aggregation"])
    attention = _discard_attention(attention, config["discard_ratio"])
    attention = _add_residual_and_normalize(attention)

    rollout = attention[:, 0]
    for layer in range(1, attention.shape[1]):
        rollout = attention[:, layer] @ rollout

    return rollout


def attention_layer(attentions: torch.Tensor, config: dict):
    """Extract attention from a single layer."""
    return _aggregate_heads(
        attentions[:, config["layer"]],
        config["head_aggregation"],
    )


def _summarize_attention(
    attention: torch.Tensor,
    constituent_patch_indices: list[list[int]],
    method: str,
) -> torch.Tensor:
    """Summarize CLS-to-patch attention for each constituent."""
    cls_to_patch_attention = attention[:, 0, 1:]
    constituent_scores = []

    for patch_indices in constituent_patch_indices:
        constituent_attention = cls_to_patch_attention[:, patch_indices]

        if method == "mean":
            score = constituent_attention.mean(dim=1)
        elif method == "max":
            score = constituent_attention.amax(dim=1)
        elif method == "min":
            score = constituent_attention.amin(dim=1)

        constituent_scores.append(score)

    return torch.stack(constituent_scores, dim=1)


# %% Embedding computation
def embedding_similarity(
    composite_embeddings: torch.Tensor,
    constituent_embeddings_by_position: list[torch.Tensor],
) -> torch.Tensor:
    """Compute composite-constituent cosine similarities for each position index."""
    return torch.stack([
        F.cosine_similarity(composite_embeddings, constituent_embeddings, dim=-1)
        for constituent_embeddings in constituent_embeddings_by_position
    ], dim=1)


# %% Metric extraction
def _build_metric_df(trial_ids, scores, metric_name, variant_id):
    """Build long-form trial-level results."""
    n_constituents = scores.shape[1]

    return pd.DataFrame({
        "Trial_id": trial_ids.repeat(n_constituents).to_numpy(),
        "Position_index": list(range(n_constituents)) * len(trial_ids),
        "Metric": metric_name,
        "Variant_id": variant_id,
        "Value": scores.detach().cpu().numpy().reshape(-1),
    })

def compute_metrics(
    trial_ids: pd.Series,
    composite_outputs: tuple,
    metrics: dict,
    constituent_embeddings_by_variant: dict,
    constituent_patch_indices: list[list[int]],
) -> pd.DataFrame:
    """Compute all metric variants and return long-form results."""
    composite_embeddings, attentions = composite_outputs
    results = []

    for metric_name, variants in metrics.items():
        for variant_id, config in variants.items():

            if metric_name == "embedding_similarity":
                scores = embedding_similarity(
                    composite_embeddings[config["embedding_function"]],
                    constituent_embeddings_by_variant[variant_id],
                )

            elif metric_name == "attention_rollout":
                attention = attention_rollout(attentions, config)
                scores = _summarize_attention(
                    attention, constituent_patch_indices, config["region_summary"]
                )

            elif metric_name == "attention_layer":
                attention = attention_layer(attentions, config)
                scores = _summarize_attention(
                    attention, constituent_patch_indices, config["region_summary"]
                )

            results.append(_build_metric_df(
                trial_ids, scores, metric_name, variant_id
            ))

    return pd.concat(results, ignore_index=True)

# %% Model loading
RANDOM_VIT_MODELS = {
    "vits16": "SIMCLR_VITS16",
    "vitb16": "SIMCLR_VITB16",
    "vitl16": "SIMCLR_VITL16",
}


def _load_random_vit(model_id: str):
    """Construct an untrained visual backbone using official SLIP constructors."""
    _, architecture, seed_id = model_id.split("_")
    seed = int(seed_id.removeprefix("seed"))

    # The model constructor performs PyTorch/timm's standard initialization.
    # Save and restore the caller's CPU RNG state.
    with torch.random.fork_rng(devices=[]):
        torch.manual_seed(seed)
        model = getattr(models, RANDOM_VIT_MODELS[architecture])(
            rand_embed=False,
            ssl_mlp_dim=4096,
            ssl_emb_dim=256,
        )

    return model


def get_device() -> torch.device:
    """Select the available computation device."""
    if torch.cuda.is_available():
        return torch.device("cuda")
    if hasattr(torch.backends, "mps") and torch.backends.mps.is_available():
        return torch.device("mps")
    return torch.device("cpu")

def adapt_model_input_size(model, canvas_size: tuple[int, int]):
    """Resize ViT positional embeddings to a new canvas size."""
    width, height = canvas_size
    vision = model.visual
    patch_embed = vision.patch_embed

    if patch_embed.img_size == (height, width):  # timm uses (H, W)
        return

    patch_h, patch_w = patch_embed.patch_size
    grid_size = (height // patch_h, width // patch_w)

    target_pos_embed = vision.pos_embed.new_empty(
        1,  # Broadcast across batches
        grid_size[0] * grid_size[1] + vision.num_tokens,
        vision.pos_embed.shape[-1],  # Embedding dimension
    )

    with torch.no_grad():
        resized = resize_pos_embed(
            vision.pos_embed,
            target_pos_embed,  # the target shape is used
            num_tokens=vision.num_tokens,
            gs_new=grid_size,
        )

    vision.pos_embed = torch.nn.Parameter(resized)
    patch_embed.img_size = (height, width)
    patch_embed.grid_size = grid_size
    patch_embed.num_patches = grid_size[0] * grid_size[1]

def load_model(
    model_id: str,
    device: torch.device,
    canvas_size: tuple[int, int] = (224, 224),
):
    """Load pretrained weights, or construct a seeded random-init ViT."""
    if model_id.startswith("random_"):
        model = _load_random_vit(model_id)
    else:
        checkpoint = torch.load(
            PROJECT_ROOT / "checkpoints" / "fb_slip" / f"{model_id}.pt",
            map_location="cpu",
        )
        args = checkpoint["args"]

        model = getattr(models, args.model)(
            rand_embed=False,
            ssl_mlp_dim=args.ssl_mlp_dim,
            ssl_emb_dim=args.ssl_emb_dim,
        )

        state_dict = {
            k.replace("module.", "", 1): v
            for k, v in checkpoint["state_dict"].items()
        }
        model.load_state_dict(state_dict, strict=True)
        del state_dict, checkpoint

    adapt_model_input_size(model, canvas_size)
    
    return model.to(device).eval()


def get_batch_size(model) -> int:
    """Select batch size based on model size to limit memory usage."""
    model_size = {
        384: "small",
        768: "base",
        1024: "large",
    }[model.vision_width]
    return BATCH_SIZES[model_size]


# %% Attention setup
def _register_attention_hooks(model):
    """Capture attention matrices from ViT layers."""
    cache = []

    def hook(module, inputs, output):
        cache.append(output.detach())

    handles = [
        block.attn.attn_drop.register_forward_hook(hook)
        for block in model.visual.blocks
    ]
    return cache, handles


def _get_patch_indices(layout: dict, patch_size: int):
    """Convert constituent regions to patch indices."""
    grid_width = layout["canvas_size"][0] // patch_size
    item_w, item_h = layout["item_size"]

    return [
        [
            row * grid_width + col
            for row in range(y // patch_size, (y + item_h) // patch_size)
            for col in range(x // patch_size, (x + item_w) // patch_size)
        ]
        for x, y in layout["positions"]
    ]


def setup_attention(model, metrics: dict, layout: dict):
    """Register attention hooks and locate constituent patches."""
    if not any(name.startswith("attention_") for name in metrics):
        return [], [], []

    attention_cache, handles = _register_attention_hooks(model)
    patch_size = model.visual.patch_embed.patch_size[0]
    constituent_patch_indices = _get_patch_indices(layout, patch_size)
    return attention_cache, handles, constituent_patch_indices


def remove_hooks(handles: list):
    """Remove registered attention hooks."""
    for handle in handles:
        handle.remove()


# %% Model forward
def forward_composites(
    model,
    composites: torch.Tensor,
    metrics: dict,
    attention_cache: list,
):
    """Forward composites once and collect raw embeddings and attention."""
    embedding_functions = {
        config["embedding_function"]
        for config in metrics.get("embedding_similarity", {}).values()
    }

    with torch.no_grad():
        visual_embeddings = model.visual(normalize_images(composites))
        embeddings = {}

        if "visual" in embedding_functions:
            embeddings["visual"] = visual_embeddings

        if "encode_image" in embedding_functions:
            embeddings["encode_image"] = (
                visual_embeddings @ model.image_projection
                if hasattr(model, "image_projection")
                else visual_embeddings  # SimCLR has no image_projection
            )

    attentions = (
        torch.stack(attention_cache, dim=1)
        if attention_cache
        else None  # No attention metrics requested; hooks not registered
    )
    attention_cache.clear()

    return embeddings, attentions

def get_image_ids(imageset_id: str, subset: str) -> list[str]:

    """Load all images in an imageset subset."""
    image_ids = get_image_ids(imageset, subset)
    image_dir = PROJECT_ROOT / "stimuli" / imageset / "images"

    image_cache = {}
    for image_id in image_ids:

# %% Stimulus construction
def _place_item(
    canvas: torch.Tensor,
    images: torch.Tensor,
    layout: dict,
    position_index: int,
) -> torch.Tensor:
    """Place a batch of images on canvases."""
    w, h = layout["item_size"]
    x, y = layout["positions"][position_index]

    items = F.interpolate(
        images,
        size=(h, w),
        mode="bilinear",
        align_corners=False,
    )

    canvas[:, :, y:y+h, x:x+w] = items
    return canvas

def make_canvas(
    batch_size: int,
    layout: dict,
    device: torch.device,
    dtype: torch.dtype,
) -> torch.Tensor:
    """Create a batch of blank canvases."""
    w, h = layout["canvas_size"]
    background = torch.tensor(
        layout["background"],
        dtype=dtype,
        device=device,
    ) / 255

    return background.view(1, 3, 1, 1).expand(
        batch_size, 3, h, w
    ).clone()

def make_constituent(
    images: torch.Tensor,
    layout: dict,
    present: str,
    position_index: int,
) -> torch.Tensor:
    """Construct constituent images in isolate or semi presentation."""
    if present == "isolate":
        return images

    canvas = make_canvas(
        len(images), layout, images.device, images.dtype
    )
    return _place_item(canvas, images, layout, position_index)
     
def make_composite(
    images: list[torch.Tensor],
    layout: dict,
) -> torch.Tensor:
    """Construct a batch of composite images."""
    canvas = make_canvas(
        len(images[0]), layout, images[0].device, images[0].dtype
    )

    for position_index, image in enumerate(images):
        _place_item(canvas, image, layout, position_index)

    return canvas

def normalize_images(images: torch.Tensor) -> torch.Tensor:
    """Normalize images using SLIP training statistics."""
    mean = images.new_tensor(SLIP_NORM_MEAN).view(1, 3, 1, 1)
    std = images.new_tensor(SLIP_NORM_STD).view(1, 3, 1, 1)
    return (images - mean) / std

# %% Extraction
def extract_constituent_embeddings(
    imageset: str,
    layout: dict,
    embedding_config: dict,
    model,
    variant_id: str,
    batch_size: int = 32,
) -> dict[str, torch.Tensor]:
    """Extract embeddings for all images and positions."""
    
    device = next(model.parameters()).device
      
    image_ids = get_image_ids(imageset, "all")
    image_dir = PROJECT_ROOT / "stimuli" / imageset / "images"

    present = embedding_config["present"]
    encoder = getattr(model, embedding_config["embedding_function"])

    n_constituents = len(layout["positions"])
    position_indices = range(n_constituents) if present == "semi" else range(1)
    
    embeddings = {}

    with torch.no_grad(), tqdm(
        total=len(image_ids),
        desc=f"Embedding: {imageset} / {variant_id}",
        unit="image",
        position=1,
        leave=False,
    ) as progress:
        for start in range(0, len(image_ids), batch_size):
            batch_ids = image_ids[start:start + batch_size]

            images = torch.stack([
                pil_to_tensor(
                    Image.open(image_dir / f"{image_id}.jpg").convert("RGB")
                ).float() / 255
                for image_id in batch_ids
            ]).to(device)

            for position_index in position_indices:
                batch = make_constituent(images, layout, present, position_index)
                features = encoder(normalize_images(batch)).cpu()
                
                for image_id, embedding in zip(batch_ids, features):
                    if present == "isolate":
                        embeddings[image_id] = embedding.unsqueeze(0)
                    else:
                        if position_index == 0:
                            embeddings[image_id] = torch.empty(
                                n_constituents, embedding.numel(),
                                dtype=embedding.dtype,
                            )
                        embeddings[image_id][position_index] = embedding

            progress.update(len(batch_ids))

    return embeddings

# %% Cache
def ensure_constituent_embeddings(
    experiment: dict,
    model,
    overwrite: bool = False,
    batch_size: int = 32,
    present: str = None,
) -> dict:
    """Load or generate constituent embeddings for the requested variants."""
    imageset = experiment["imageset"]
    model_id = experiment["model"]
    layout_id = experiment["layout_id"]
    layout = experiment["layout"]
    device = next(model.parameters()).device

    embeddings_by_variant = {}
    embedding_configs = experiment["metrics"].get("embedding_similarity", {})

    selected_variants = {
        variant_id: config
        for variant_id, config in embedding_configs.items()
        if present is None or config["present"] == present
    }

    for variant_id, config in selected_variants.items():

        cache_dir = (
            PROJECT_ROOT / "data" / "constituent_embeddings"
            / imageset / model_id
        )
        if config["present"] == "isolate":
            path = cache_dir / "shared" / f"{variant_id}.pt"
        else:
            path = cache_dir / layout_id / f"{variant_id}.pt"

        if path.exists() and not overwrite:
            message(f"Loading constituent embeddings: {path}")
            embeddings = torch.load(path, map_location="cpu")
        else:
            embeddings = extract_constituent_embeddings(
                imageset, layout, config, model,
                variant_id=variant_id,
                batch_size=batch_size,
            )
            path.parent.mkdir(parents=True, exist_ok=True)
            torch.save(embeddings, path)
            message(f"Generated constituent embeddings: {path}")

        if config["present"] == "isolate":
            n_constituents = len(layout["positions"])
            embeddings = {
                k: v.repeat(n_constituents, 1)
                for k, v in embeddings.items()
            }

        embeddings_by_variant[variant_id] = {
            k: v.to(device) for k, v in embeddings.items()
        }

    return embeddings_by_variant

# %% Imports
import multiprocessing as mp
import os
from datetime import timedelta
from time import perf_counter

from tqdm import tqdm

from config import (
    ACTIVE_SWEEP,
    EXPERIMENT_SWEEPS,
    MAX_ATTEMPTS,
    PROJECT_ROOT,
    message,
    sweep_experiments,
)

# %% Sweep resolution
def _get_sweep_ids() -> list[str]:
    """Resolve the active sweep selection."""
    if ACTIVE_SWEEP == "all":
        return list(EXPERIMENT_SWEEPS)
    elif isinstance(ACTIVE_SWEEP, str):
        return [ACTIVE_SWEEP]
    elif isinstance(ACTIVE_SWEEP, list):
        return ACTIVE_SWEEP

# %% Experiment execution
def _ignore_warning():
    import warnings

    warnings.filterwarnings(
        "ignore",
        message=r"pkg_resources is deprecated as an API.*",
        category=UserWarning,
    )
    
def _run_experiment(experiment: dict, progress_lock):
    """Run an experiment in an isolated worker process."""
    tqdm.set_lock(progress_lock)
    _ignore_warning()

    message(f"Worker PID: {os.getpid()}", flush=True)
    
    from script.experiment.experiment import run_experiment
    run_experiment(experiment)

def _run_with_retries(experiment: dict, context, progress_lock):
    """Run an experiment with retries and record its completion time."""
    result_path = (
        PROJECT_ROOT / "results"
        / experiment["protocol_id"] / experiment["experiment_id"]
        / "results.csv"
    )
    already_completed = result_path.exists()
    start_time = perf_counter()

    for attempt in range(1, MAX_ATTEMPTS + 1):
        process = context.Process(
            target=_run_experiment,
            args=(experiment, progress_lock),
        )
        process.start()
        process.join()

        if process.exitcode == 0:
            if not already_completed:
                elapsed = timedelta(seconds=int(perf_counter() - start_time))
                tqdm.write(f"Completed: {experiment['experiment_id']} | {elapsed}")
            return

        if attempt == MAX_ATTEMPTS:
            raise RuntimeError(
                f"Experiment failed after {MAX_ATTEMPTS} attempts: "
                f"{experiment['experiment_id']} "
                f"(exit code: {process.exitcode})"
            )

        tqdm.write(
            f"Retrying {experiment['experiment_id']} "
            f"({attempt + 1}/{MAX_ATTEMPTS}) "
            f"after worker exit code {process.exitcode}"
        )


# %% Main
def main():
    """Run all selected sweeps."""
    context = mp.get_context("spawn")
    progress_lock = context.RLock()
    tqdm.set_lock(progress_lock)

    for sweep_id in _get_sweep_ids():
        experiments = list(sweep_experiments(sweep_id))
    
        for experiment in tqdm(
            experiments,
            desc=f"{sweep_id}",
            unit="experiment",
            position=0,
            leave=True,
        ):
            _run_with_retries(experiment, context, progress_lock)


if __name__ == "__main__":
    main()

# %% Image processing
def _crop_black_borders(image: Image.Image) -> Image.Image:
    """Crop black borders using the nonblack bounding box."""
    bbox = image.convert("L").getbbox()
    return image.crop(bbox) if bbox else image


def standardize_images(imageset: str):
    """Process raw images into standardized 224x224 JPG files."""
    raw_dir = PROJECT_ROOT / "stimuli" / imageset / "raw" / "images"
    output_dir = PROJECT_ROOT / "stimuli" / imageset / "images"
    output_dir.mkdir(parents=True, exist_ok=True)

    image_paths = sorted(
        path for path in raw_dir.rglob("*")
        if path.suffix.lower() in IMAGE_EXTENSIONS
    )

    for path in tqdm(image_paths, desc=f"Standardizing {imageset}", unit="image"):
        with Image.open(path) as image:
            if "crop_black_borders" in STANDARDIZE_PIPELINE.get(imageset, []):
                image = _crop_black_borders(image)
            image = image.resize(STANDARD_IMAGE_SIZE, Image.Resampling.LANCZOS)
            image = image.convert("RGB")
            image.save(output_dir / f"{path.stem}.jpg", "JPEG", quality=95)

    """Clean the EmoMadrid ratings file."""

        raw_path,
        sep=";",
        decimal=",",
        skiprows=1,
        encoding="cp1252",
        dtype={"EM CODE": str},

    df.columns = df.columns.str.replace(r"\s+", " ", regex=True).str.strip()
    df = df.rename(columns={
        "EM CODE": "Image_id",
        "Valence mean": "Valence",
        "Arousal mean": "Arousal",
    })
    df = df.drop(columns=["Thumbnail"])
    df.columns = df.columns.str.replace(" ", "_", regex=False)
    df["Image_id"] = df["Image_id"].str.strip()
    return df


def clean_naps_ratings() -> pd.DataFrame:
    """Load the NAPS ratings file without cleaning."""
    raw_path = PROJECT_ROOT / "stimuli" / "NAPS" / "ratings.csv"

# %% Category subsets
def _update_category_subsets(imageset: str, ratings: pd.DataFrame):
    """Regenerate category columns while preserving other subset columns."""
    path = PROJECT_ROOT / "stimuli" / imageset / "subsets.csv"
    subsets = ratings[["Image_id"]].copy()

    if "Category" in ratings:
        category_columns = pd.get_dummies(
            ratings["Category"], prefix="category", prefix_sep="__", dtype=int
        )
        subsets = pd.concat([subsets, category_columns], axis=1)

    if path.exists():
        previous = pd.read_csv(path, dtype={"Image_id": str}).set_index("Image_id")
        preserved_columns = [
            column for column in previous
            if not column.startswith("category__")
        ]
        for column in preserved_columns:
            subsets[column] = (
                previous[column]
                .reindex(subsets["Image_id"])
                .fillna(0)
                .astype(int)
                .to_numpy()
            )

    subsets.to_csv(path, index=False)

# ABA_VLM preprocessing

Standardize raw affective image datasets for experiments. Run from the **project root**.

## Files

| Script | Responsibility |
| --- | --- |
| `image_standardization.py` | Collect raw images, optionally crop black borders, resize to 224 Ã— 224, save JPGs. |
| `rating_cleaning.py` | Clean OASIS/EmoMadrid ratings; generate `category__*` columns in `subsets.csv` without overwriting manually maintained columns. |
| `visual_feature_extraction.py` | Extract image-level grayscale, JPEG, CIELAB, and RGB statistics. |
| `run_preprocessing.py` | Run all three steps in sequence. |

## Directory layout

```text
stimuli/
  OASIS/
    raw/
      images/                  # source images (nested folders are supported)
      oasis.csv                # original OASIS ratings
    images/                   # standardized JPGs
    ratings.csv
    subsets.csv
    visual_features.csv
  EmoMadrid/
    raw/
      images/
      1_EMindex_2025_07_07(EMindex).csv
    images/
    ratings.csv
    subsets.csv
    visual_features.csv
```

## Run

```bash
python -m script.preprocessing.run_preprocessing OASIS
python -m script.preprocessing.run_preprocessing EmoMadrid
```

Individual steps can be run with the same module pattern:

```bash
python -m script.preprocessing.image_standardization OASIS
python -m script.preprocessing.rating_cleaning OASIS
python -m script.preprocessing.visual_feature_extraction OASIS
```

The ratings cleaning currently implements the **OASIS** and **EmoMadrid** source formats from the project's earlier preprocessing scripts. Adding NAPS requires its own source-specific ratings cleaning function. Image standardization resizes both datasets to `STANDARD_IMAGE_SIZE` from `config`, currently `(224, 224)`. The EmoMadrid procedure also crops the nonblack bounding box before resizing, matching the earlier preprocessing logic.

## Standardized outputs

- `images/<Image_id>.jpg`: RGB images at the standard size, JPEG quality 95.
- `ratings.csv`: original metadata retained with standardized `Image_id`, `Valence`, `Arousal`, and `Category` when present in the source.
- `subsets.csv`: `Image_id` followed by binary subset membership columns.
- `visual_features.csv`: `Image_id`, `Luminance`, `Contrast`, `JpegSize`, `Entropy`, `Ldim_mean`, `Adim_mean`, `Bdim_mean`, `Rch_mean`, `Gch_mean`, `Bch_mean`.

Visual features retain the definitions from the corrected feature-extraction script: grayscale mean and standard deviation, grayscale Shannon entropy, JPEG byte size at quality 80, mean D65 CIELAB coordinates, and mean RGB channels.

## Subset management

Each `subsets.csv` has one row per image, identified by `Image_id`.

```csv
Image_id,category__Animal,category__People,manual__Textureless
A,1,0,0
B,0,1,1
C,1,0,1
```

`category__*` columns are automatically regenerated from the `Category` column of `ratings.csv`. If that column is absent, no category columns are generated. Other columns (for example, `manual__Textureless` or `manual__NSFW`) are retained when preprocessing is rerun, matched by `Image_id` rather than row order. Newly introduced image IDs receive `0` for existing manual subset columns. The reserved subset name `all` means all image IDs and does not require a separate column.

For **experiment code**, update `get_image_ids()` in `script/experiment/stimulus.py` to read `subsets.csv` rather than `ratings.csv`:

```python
def get_image_ids(imageset_id: str, subset: str) -> list[str]:
    path = PROJECT_ROOT / "stimuli" / imageset_id / "subsets.csv"
    subsets = pd.read_csv(path, dtype={"Image_id": str})
    if subset == "all":
        image_ids = subsets["Image_id"]
    else:
        image_ids = subsets.loc[subsets[subset] == 1, "Image_id"]
    return sorted(image_ids.tolist())
```

When subset membership changes, regenerate any previously cached design matrices that were constructed from that subset name (or use a versioned subset name).

"""Extract visual statistics from standardized images."""

# %% Imports
import io

import numpy as np
import pandas as pd
from PIL import Image
from tqdm import tqdm

from config import PROJECT_ROOT


# %% Color conversion
def _rgb_to_lab(rgb_arr: np.ndarray) -> np.ndarray:
    """Convert sRGB pixel values (0-255) to CIELAB using D65 white point."""
    rgb = rgb_arr.astype(np.float64) / 255.0
    rgb_linear = np.where(
        rgb <= 0.04045,
        rgb / 12.92,
        ((rgb + 0.055) / 1.055) ** 2.4,
    )

    rgb_to_xyz = np.array([
        [0.4124564, 0.3575761, 0.1804375],
        [0.2126729, 0.7151522, 0.0721750],
        [0.0193339, 0.1191920, 0.9503041],
    ])
    xyz = rgb_linear @ rgb_to_xyz.T
    xyz_scaled = xyz / np.array([0.95047, 1.00000, 1.08883])

    delta = 6 / 29
    f = np.where(
        xyz_scaled > delta ** 3,
        np.cbrt(xyz_scaled),
        xyz_scaled / (3 * delta ** 2) + 4 / 29,
    )
    L = 116 * f[:, :, 1] - 16
    a = 500 * (f[:, :, 0] - f[:, :, 1])
    b = 200 * (f[:, :, 1] - f[:, :, 2])
    return np.stack([L, a, b], axis=2)


# %% Image statistics
def extract_image_physical_properties(image_path) -> dict:
    """Compute grayscale, JPEG, CIELAB, and RGB image properties."""
    with Image.open(image_path) as image:
        img = image.convert("RGB")

    gray_arr = np.asarray(img.convert("L"), dtype=np.float64)
    luminance = float(gray_arr.mean())
    contrast = float(gray_arr.std())

    buffer = io.BytesIO()
    img.save(buffer, format="JPEG", quality=80)
    jpeg_size = buffer.tell()

    hist, _ = np.histogram(gray_arr.ravel(), bins=256, range=(0, 256))
    prob = hist / hist.sum()
    nonzero = prob[prob > 0]
    entropy = float(-np.sum(nonzero * np.log2(nonzero)))

    rgb_arr = np.asarray(img, dtype=np.float64)
    lab_arr = _rgb_to_lab(rgb_arr)

    return {
        "Luminance": luminance,
        "Contrast": contrast,
        "JpegSize": jpeg_size,
        "Entropy": entropy,
        "Ldim_mean": float(lab_arr[:, :, 0].mean()),
        "Adim_mean": float(lab_arr[:, :, 1].mean()),
        "Bdim_mean": float(lab_arr[:, :, 2].mean()),
        "Rch_mean": float(rgb_arr[:, :, 0].mean()),
        "Gch_mean": float(rgb_arr[:, :, 1].mean()),
        "Bch_mean": float(rgb_arr[:, :, 2].mean()),
    }


# %% Feature extraction
def extract_visual_features(imageset: str):
    """Write image-level visual statistics to visual_features.csv."""
    image_dir = PROJECT_ROOT / "stimuli" / imageset / "images"
    image_paths = sorted(image_dir.glob("*.jpg"))

    rows = []
    for path in tqdm(image_paths, desc=f"Visual features: {imageset}", unit="image"):
        rows.append({"Image_id": path.stem, **extract_image_physical_properties(path)})

    output_path = PROJECT_ROOT / "stimuli" / imageset / "visual_features.csv"
    pd.DataFrame(rows).to_csv(output_path, index=False)


# %% Run
if __name__ == "__main__":
    import argparse

    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("imageset")
    args = parser.parse_args()
    extract_visual_features(args.imageset)

# Coefficient plots: layouts as rows, imagesets as columns.
# Random, SimCLR, SLIP and CLIP are compared on the same axes.

plot_coefficients <- function(coefficients, sweep_ids, priority_mode,
                              layout_levels, imageset_levels, predictors,
                              metric, variant_id) {
  library(ggplot2)

  df <- data.table::copy(coefficients[
    Sweep %in% sweep_ids &
      Priority_mode == priority_mode &
      Predictor %in% predictors &
      Metric == metric & Variant_id == variant_id
  ])
  if (nrow(df) == 0) return(NULL)

  df[, Layout := factor(Layout, levels = layout_levels)]
  df[, Imageset := factor(Imageset, levels = imageset_levels)]
  df[, Predictor := factor(Predictor, levels = predictors)]
  df[, Training := factor(
    Training, levels = c("random", "simclr", "slip", "clip"),
    labels = c("Random", "SimCLR", "SLIP", "CLIP")
  )]
  df[, Size := factor(
    Size, levels = c("small", "base", "large"),
    labels = c("Small", "Base", "Large")
  )]

  # Four training groups, each with three size positions.
  predictor_x <- setNames(seq_along(predictors), predictors)
  model_offset <- c(Random = -0.27, SimCLR = -0.09, SLIP = 0.09, CLIP = 0.27)
  size_offset <- c(Small = -0.055, Base = 0, Large = 0.055)
  df[, X := predictor_x[as.character(Predictor)] +
            model_offset[as.character(Training)] +
            size_offset[as.character(Size)]]

  df[, Fill := ifelse(Significant %in% TRUE, as.character(Training), "n.s.")]
  model_colors <- c(Random = "#777777", SimCLR = "#D55E00",
                    SLIP = "#0072B2", CLIP = "#009E73")

  ggplot(df, aes(x = X, y = Beta, colour = Training, shape = Size, fill = Fill)) +
    geom_hline(yintercept = 0, linetype = "dashed", colour = "grey65", linewidth = 0.35) +
    geom_point(size = 2.5, stroke = 0.8, na.rm = TRUE) +
    facet_grid(Layout ~ Imageset, drop = FALSE) +
    scale_x_continuous(
      breaks = seq_along(predictors), labels = predictors,
      limits = c(0.55, length(predictors) + 0.45)
    ) +
    scale_colour_manual(name = "Model", values = model_colors) +
    scale_shape_manual(name = "Size", values = c(Small = 21, Base = 24, Large = 22)) +
    scale_fill_manual(values = c(model_colors, "n.s." = "white"), guide = "none") +
    guides(
      shape = guide_legend(order = 1, override.aes = list(fill = "white", colour = "black")),
      colour = guide_legend(
        order = 2,
        override.aes = list(
          shape = 21,
          fill = unname(model_colors[levels(droplevels(df$Training))])
        )
      )
    ) +
    labs(
      title = paste(variant_id, "—", priority_mode, "priority"),
      x = NULL, y = "Standardized regression coefficient (β)",
      caption = "Open symbols: n.s.  |  Filled symbols: adjusted p < .05"
    ) +
    theme_bw(base_size = 11) +
    theme(
      panel.grid.minor = element_blank(),
      panel.grid.major.x = element_blank(),
      strip.background = element_rect(fill = "grey93", colour = "white"),
      strip.text.x = element_text(face = "bold", size = 10),
      strip.text.y = element_text(angle = 0, size = 8),
      axis.text.x = element_text(size = 9),
      legend.position = "bottom",
      legend.box = "horizontal",
      plot.caption = element_text(hjust = 1, size = 9)
    )
}
